# Swift 种子语料 — fuzz 变异起点（fuzz_input.py --lang swift）

import Foundation

struct Item: Codable {
    let name: String
    let count: Int
}

func process(_ data: Data) throws {
    let items = try JSONDecoder().decode([Item].self, from: data)
    for item in items {
        print(item.name, item.count * 2)
    }
}

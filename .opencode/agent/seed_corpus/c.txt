#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(void) {
    char buf[256];
    if (fgets(buf, sizeof(buf), stdin) == NULL) return 1;
    int n = atoi(buf);
    for (int i = 0; i < n; i++) {
        printf("%d\n", i);
    }
    char *p = malloc(1024);
    strcpy(p, buf);
    free(p);
    return 0;
}
